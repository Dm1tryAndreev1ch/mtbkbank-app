## sendBroadcast Helper Final Shape

```js
// backend/src/push/index.js
async function sendBroadcast(recipients, { title, body, data }) {
  const messages = (recipients || [])
    .filter((r) => r && r.expoPushToken && Expo.isExpoPushToken(r.expoPushToken))
    .map((r) => ({ to: r.expoPushToken, sound: 'default', title, body, data: data || {} }));
  if (messages.length === 0) return { ok: 0, error: 0 };
  const chunks = expo.chunkPushNotifications(messages);
  let ok = 0, error = 0;
  for (const chunk of chunks) {
    try {
      const tickets = await expo.sendPushNotificationsAsync(chunk);
      for (const t of tickets) {
        if (t && t.status === 'ok') ok += 1; else error += 1;
      }
    } catch (e) {
      logger.error({ err: e }, 'Push broadcast chunk send failed');
      error += chunk.length;
    }
  }
  return { ok, error };
}
```

Filters out tokens that fail `Expo.isExpoPushToken` (silent skip — admin has no actionable signal for malformed user tokens). Per-chunk send failures bump `error` by `chunk.length` so partial outages are observable.

## Trade Cancel Pattern (IDOR-safe)

```js
router.post('/:id/cancel', reqValidator(adminTradeCancelSchema), async (req, res, next) => {
  const id = req.params.id; // URL only — never req.body
  // ...withAudit closure
  const before = await tx.cardTrade.findUnique({ where: { id }, select: { id, status, fromUserId, toUserId }});
  if (!before) throw new AppError('NOT_FOUND', 404);
  if (before.status !== 'PENDING') throw new AppError('TRADE_NOT_CANCELLABLE', 409);
  const after = await tx.cardTrade.update({ where: { id }, data: { status: 'CANCELLED' }, ... });
  setAudit({ before, after, reason });
});
```

`adminTradeCancelSchema` only contains `{ reason: z.string().min(3).max(500).optional() }` — no `id` / `tradeId` field. The body is therefore not a vehicle for IDOR.
